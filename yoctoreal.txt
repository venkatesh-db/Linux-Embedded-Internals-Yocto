#include <iostream>
#include <fstream>
#include <chrono>
#include <thread>

int main()
{
    int cpuTemp = 45;

    while (true)
    {
        std::ofstream sensorFile("/run/hwmon-temperature");

        sensorFile << cpuTemp << std::endl;

        sensorFile.close();

        std::cout << "[hwmon-service] cpu_temp=" << cpuTemp << std::endl;

        cpuTemp++;

        if (cpuTemp > 70)
        {
            cpuTemp = 45;
        }

        std::this_thread::sleep_for(std::chrono::seconds(2));
    }

    return 0;
}

EOF

Restart=always
RestartSec=3

[Install]

EOF

#include <iostream>
#include <fstream>
#include <string>
#include <chrono>
#include <thread>

int main()
{
    while (true)
    {
        std::ifstream sensorFile("/run/hwmon-temperature");

        std::string temperature;

        getline(sensorFile, temperature);

        sensorFile.close();

        std::ofstream telemetry("/run/telemetry-data");

        telemetry << "CPU_TEMP=" << temperature << std::endl;

        telemetry.close();

        std::cout << "[telemetry-service] telemetry updated temp="
                  << temperature << std::endl;

        std::this_thread::sleep_for(std::chrono::seconds(2));
    }

    return 0;
}

EOF
