
# ============================================================
# STEP 1 — Go to Poky Directory
# ============================================================

cd ~/poky


# ============================================================
# STEP 2 — Setup Yocto Environment
# ============================================================

source oe-init-build-env


# ============================================================
# STEP 3 — Verify Custom Layer
# ============================================================

bitbake-layers show-layers


# ============================================================
# STEP 4 — Create RootFS Recipe Directory
# ============================================================

mkdir -p ../meta-ilo/recipes-core/custom-rootfs/files


# ============================================================
# STEP 5 — Create Custom Welcome Banner
# ============================================================

cat > ../meta-ilo/recipes-core/custom-rootfs/files/motd << 'EOF'

==========================================
      Jai Ganesha Embedded Linux
==========================================

Welcome to Custom Yocto RootFS

EOF


# ============================================================
# STEP 6 — Create Startup Script
# ============================================================

cat > ../meta-ilo/recipes-core/custom-rootfs/files/startup.sh << 'EOF'

#!/bin/sh

echo "Custom startup script running..."

date > /tmp/startup-time.log

EOF


# ============================================================
# STEP 7 — Give Execute Permission
# ============================================================

chmod +x ../meta-ilo/recipes-core/custom-rootfs/files/startup.sh


# ============================================================
# STEP 8 — Create RootFS Recipe
# ============================================================

cat > ../meta-ilo/recipes-core/custom-rootfs/custom-rootfs_1.0.bb << 'EOF'

DESCRIPTION = "Custom RootFS Files"

LICENSE = "MIT"

LIC_FILES_CHKSUM = "file://${COMMON_LICENSE_DIR}/MIT;md5=0835ade698e0bcf8506ecda2f7b4f302"

SRC_URI = " \
    file://motd \
    file://startup.sh \
"

S = "${WORKDIR}"

do_install() {

    install -d ${D}${sysconfdir}

    install -m 0644 motd \
        ${D}${sysconfdir}/motd

    install -d ${D}${bindir}

    install -m 0755 startup.sh \
        ${D}${bindir}/startup.sh
}

FILES:${PN} += " \
    ${sysconfdir}/motd \
    ${bindir}/startup.sh \
"

EOF


# ============================================================
# STEP 9 — Add Package Into Image
# ============================================================

echo 'IMAGE_INSTALL:append = " custom-rootfs"' >> conf/local.conf


# ============================================================
# STEP 10 — Build Image
# ============================================================

bitbake core-image-minimal


# ============================================================
# STEP 11 — Run QEMU
# ============================================================

runqemu qemux86-64 nographic


# ============================================================
# STEP 12 — Verify RootFS Customization
# ============================================================

cat /etc/motd


# ============================================================
# STEP 13 — Verify Startup Script
# ============================================================

/usr/bin/startup.sh


# ============================================================
# STEP 14 — Verify Generated File
# ============================================================

cat /tmp/startup-time.log


# ============================================================
# OPTIONAL DEBUG COMMANDS
# ============================================================

# Verify file exists
ls /etc/motd

# Verify script exists
ls /usr/bin/startup.sh

# Check permissions
ls -l /usr/bin/startup.sh

# Search installed package files
find / -name startup.sh
