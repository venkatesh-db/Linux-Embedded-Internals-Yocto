cat > process_demo.c << 'EOF'

#include <stdio.h>
#include <unistd.h>
#include <sys/types.h>

int main()
{
    pid_t pid;

    pid = fork();

    if (pid == 0)
    {
        while (1)
        {
            printf("Child Process Running PID=%d\n", getpid());

            sleep(2);
        }
    }
    else
    {
        while (1)
        {
            printf("Parent Process Running PID=%d\n", getpid());

            sleep(2);
        }
    }

    return 0;
}

EOF

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

int main()
{
    while (1)
    {
        char* buffer;

        buffer = malloc(1024 * 1024);

        printf("Allocated 1MB memory at %p\n", buffer);

        sleep(2);
    }

    return 0;
}

EOF

# =====================================================================

# =====================================================================
# CREATE DRIVER SOURCE
# =====================================================================

cat > simple_driver.c << 'EOF'

#include <linux/module.h>
#include <linux/kernel.h>

static int __init simple_driver_init(void)
{
    printk(KERN_INFO "Simple Driver Loaded\n");

    return 0;
}

static void __exit simple_driver_exit(void)
{
    printk(KERN_INFO "Simple Driver Removed\n");
}

module_init(simple_driver_init);

module_exit(simple_driver_exit);

MODULE_LICENSE("GPL");

EOF



# =====================================================================
# CREATE DRIVER MAKEFILE
# =====================================================================

cat > Makefile << 'EOF'

obj-m += simple_driver.o

KDIR := /lib/modules/$(shell uname -r)/build

PWD := $(shell pwd)

all:
	make -C $(KDIR) M=$(PWD) modules

clean:
	make -C $(KDIR) M=$(PWD) clean

EOF



# =====================================================================
# BUILD DRIVER
# =====================================================================

make
