
# ============================================================
# STEP 1 — Go To Poky Directory
# ============================================================

cd ~/poky


# ============================================================
# STEP 2 — Setup Yocto Environment
# ============================================================

source oe-init-build-env


# ============================================================
# STEP 3 — Build Any Package
# ============================================================

bitbake core-image-minimal


# ============================================================
# STEP 4 — View Full BitBake Task Logs
# ============================================================

ls tmp/work/


# ============================================================
# STEP 5 — Find Recipe Work Directory
# ============================================================

find tmp/work -name "temp"


# ============================================================
# STEP 6 — Open Build Logs
# ============================================================

less tmp/work/*/*/*/temp/log.do_compile


# ============================================================
# STEP 7 — Open Install Logs
# ============================================================

less tmp/work/*/*/*/temp/log.do_install


# ============================================================
# STEP 8 — Open Configure Logs
# ============================================================

less tmp/work/*/*/*/temp/log.do_configure


# ============================================================
# STEP 9 — Open Packaging Logs
# ============================================================

less tmp/work/*/*/*/temp/log.do_package


# ============================================================
# STEP 10 — Search ERROR Lines
# ============================================================

grep ERROR tmp/work/*/*/*/temp/log.*


# ============================================================
# STEP 11 — Search WARNING Lines
# ============================================================

grep WARNING tmp/work/*/*/*/temp/log.*


# ============================================================
# STEP 12 — Follow Live Build Logs
# ============================================================

tail -f tmp/work/*/*/*/temp/log.do_compile

# View BitBake environment
bitbake -e core-image-minimal

# View all tasks
bitbake -c listtasks core-image-minimal

# Verbose build
bitbake -v core-image-minimal

bitbake -DDD core-image-minimal

bitbake -k core-image-minimal
