cat > ../meta-ilo/recipes-core/custom-rootfs/custom-rootfs_1.0.bb << 'EOF'

DESCRIPTION = "Custom RootFS Files"

LICENSE = "MIT"

LIC_FILES_CHKSUM = "file://${COMMON_LICENSE_DIR}/MIT;md5=0835ade698e0bcf8506ecda2f7b4f302"

SRC_URI = " \
    file://motd \
    file://startup.sh \
"

S = "${WORKDIR}"

do_install() {

    install -d ${D}${sysconfdir}

    install -m 0644 motd \
        ${D}${sysconfdir}/motd

    install -d ${D}${bindir}

    install -m 0755 startup.sh \
        ${D}${bindir}/startup.sh
}

FILES:${PN} += " \
    ${sysconfdir}/motd \
    ${bindir}/startup.sh \
"

EOF


echo 'IMAGE_INSTALL:append = " custom-rootfs"' \
>> conf/local.conf


# ============================================================
# STEP 1 — Create Recipe Directory
# ============================================================

mkdir -p \
../meta-ilo/recipes-core/sensor-monitor/files


cat > \
../meta-ilo/recipes-core/sensor-monitor/files/sensor-monitor.cpp << 'EOF'

#include <iostream>
#include <unistd.h>

int main()
{
    while (true)
    {
        std::cout << "Sensor monitor running" << std::endl;

        sleep(5);
    }

    return 0;
}

EOF


cat > \
../meta-ilo/recipes-core/sensor-monitor/files/sensor-monitor.service << 'EOF'

[Unit]
Description=Sensor Monitor Service
After=network.target

[Service]
Type=simple
ExecStart=/usr/bin/sensor-monitor
Restart=always

[Install]
WantedBy=multi-user.target

EOF


cat > \
../meta-ilo/recipes-core/sensor-monitor/sensor-monitor_1.0.bb << 'EOF'

DESCRIPTION = "Sensor Monitor Service"

LICENSE = "MIT"

LIC_FILES_CHKSUM = "file://${COMMON_LICENSE_DIR}/MIT;md5=0835ade698e0bcf8506ecda2f7b4f302"

SRC_URI = " \
    file://sensor-monitor.cpp \
    file://sensor-monitor.service \
"

S = "${WORKDIR}"

inherit systemd

SYSTEMD_SERVICE:${PN} = "sensor-monitor.service"

SYSTEMD_AUTO_ENABLE = "enable"

do_compile() {

    ${CXX} ${CXXFLAGS} ${LDFLAGS} \
        sensor-monitor.cpp \
        -o sensor-monitor
}

do_install() {

    install -d ${D}${bindir}

    install -m 0755 sensor-monitor \
        ${D}${bindir}/sensor-monitor

    install -d ${D}${systemd_system_unitdir}

    install -m 0644 \
        ${WORKDIR}/sensor-monitor.service \
        ${D}${systemd_system_unitdir}/sensor-monitor.service
}

FILES:${PN} += " \
    ${bindir}/sensor-monitor \
    ${systemd_system_unitdir}/sensor-monitor.service \
"

EOF


echo 'IMAGE_INSTALL:append = " sensor-monitor"' \
>> conf/local.conf
