
# ============================================================
# STEP 1 — Go To Poky Directory
# ============================================================

cd ~/poky


# ============================================================
# STEP 2 — Setup Yocto Environment
# ============================================================

source oe-init-build-env


# ============================================================
# STEP 3 — Configure Parallel Threads
# ============================================================

vi conf/local.conf


# ============================================================
# ADD BELOW
# ============================================================

BB_NUMBER_THREADS = "8"

PARALLEL_MAKE = "-j8"


# ============================================================
# STEP 4 — Build Image
# ============================================================

bitbake core-image-minimal


# ============================================================
# STEP 5 — Monitor CPU Usage
# ============================================================

top


# ============================================================
# STEP 6 — Monitor Memory Usage
# ============================================================

free -h


# ============================================================
# STEP 7 — Monitor Disk Usage
# ============================================================

df -h


# ============================================================
# STEP 8 — Check Build Time
# ============================================================

time bitbake core-image-minimal


# ============================================================
# STEP 9 — Enable Build History
# ============================================================

echo 'INHERIT += "buildhistory"' >> conf/local.conf


# ============================================================
# STEP 10 — Rebuild
# ============================================================

bitbake core-image-minimal


# ============================================================
# STEP 11 — View Build History
# ============================================================

ls buildhistory/


# ============================================================
# STEP 12 — Enable Build Statistics
# ============================================================

echo 'USER_CLASSES += "buildstats"' >> conf/local.conf


# ============================================================
# STEP 13 — Rebuild
# ============================================================

bitbake core-image-minimal


# ============================================================
# STEP 14 — View Build Stats
# ============================================================

ls tmp/buildstats/


# ============================================================
# STEP 15 — Analyze Slow Tasks
# ============================================================

find tmp/buildstats -name "*.log"


# ============================================================
# OPTIONAL PERFORMANCE COMMANDS
# ============================================================

# Reuse shared state cache
echo 'SSTATE_DIR ?= "/opt/yocto/sstate-cache"' >> conf/local.conf

# Shared downloads directory
echo 'DL_DIR ?= "/opt/yocto/downloads"' >> conf/local.conf

# Build single package only
bitbake sensor-monitor

# Skip unnecessary recommendations
echo 'BAD_RECOMMENDATIONS += "packagegroup-core-x11"' >> conf/local.conf
