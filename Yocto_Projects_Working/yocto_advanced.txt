#include <iostream>
#include <fstream>
#include <unistd.h>

int main()
{
    int temp = 40;

    while (true)
    {
        std::ofstream file("/tmp/sensor-data");

        file << temp << std::endl;

        file.close();

        std::cout << "[sensor-monitor] temp=" << temp << std::endl;

        temp++;

        if (temp > 60)
        {
            temp = 40;
        }

        sleep(3);
    }

    return 0;
}

EOF



cat > ../meta-ilo/recipes-core/sensor-monitor/files/sensor-monitor.service << 'EOF'

[Unit]
Description=Sensor Monitor Service
After=network.target

[Service]
Type=simple
ExecStart=/usr/bin/sensor-monitor
Restart=always
RestartSec=2

[Install]
WantedBy=multi-user.target

EOF

LICENSE = "MIT"

LIC_FILES_CHKSUM = "file://${COMMON_LICENSE_DIR}/MIT;md5=0835ade698e0bcf8506ecda2f7b4f302"

SRC_URI = " \
    file://sensor-monitor.cpp \
    file://sensor-monitor.service \
"

S = "${WORKDIR}"

inherit systemd

SYSTEMD_SERVICE:${PN} = "sensor-monitor.service"

do_compile() {

    ${CXX} ${CXXFLAGS} sensor-monitor.cpp -o sensor-monitor

}

do_install() {

    install -d ${D}${bindir}

    install -m 0755 sensor-monitor \
        ${D}${bindir}/sensor-monitor

    install -d ${D}${systemd_system_unitdir}

    install -m 0644 \
        ${WORKDIR}/sensor-monitor.service \
        ${D}${systemd_system_unitdir}/sensor-monitor.service
}

FILES:${PN} += " \
    ${bindir}/sensor-monitor \
    ${systemd_system_unitdir}/sensor-monitor.service \
"

EOF

#include <iostream>
#include <fstream>
#include <string>
#include <unistd.h>

int main()
{
    while (true)
    {
        std::ifstream sensor("/tmp/sensor-data");

        std::string temp;

        getline(sensor, temp);

        sensor.close();

        std::ofstream dbus("/tmp/dbus-temperature");

        dbus << "DBUS_TEMP=" << temp << std::endl;

        dbus.close();

        std::cout << "[dbus-publisher] published temp=" << temp << std::endl;

        sleep(3);
    }

    return 0;
}

EOF

[Service]

Restart=always

[Install]
WantedBy=multi-user.target

EOF
