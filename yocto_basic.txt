
```bash
# ============================================================
# STEP 1 — Go to Poky Directory
# ============================================================

cd ~/poky


# ============================================================
# STEP 2 — Setup Yocto Environment
# ============================================================

source oe-init-build-env


# ============================================================
# STEP 3 — Create Custom Layer
# ============================================================

bitbake-layers create-layer ../meta-ilo


# ============================================================
# STEP 4 — Add Layer
# ============================================================

bitbake-layers add-layer ../meta-ilo


# ============================================================
# STEP 5 — Verify Layer
# ============================================================

bitbake-layers show-layers


# ============================================================
# STEP 6 — Create Recipe Directory Structure
# ============================================================

mkdir -p ../meta-ilo/recipes-core/sensor-monitor/files


# ============================================================
# STEP 7 — Create Application Source File
# ============================================================

cat > ../meta-ilo/recipes-core/sensor-monitor/files/sensor-monitor.cpp << 'EOF'
#include <iostream>
#include <unistd.h>

int main()
{
    while (true)
    {
        std::cout << "Sensor monitor running" << std::endl;
        sleep(5);
    }

    return 0;
}
EOF


# ============================================================
# STEP 8 — Create systemd Service File
# ============================================================

cat > ../meta-ilo/recipes-core/sensor-monitor/files/sensor-monitor.service << 'EOF'
[Unit]
Description=Sensor Monitor Service
After=network.target

[Service]
Type=simple
ExecStart=/usr/bin/sensor-monitor
Restart=always

[Install]
WantedBy=multi-user.target
EOF


# ============================================================
# STEP 9 — Create BitBake Recipe
# ============================================================

cat > ../meta-ilo/recipes-core/sensor-monitor/sensor-monitor_1.0.bb << 'EOF'

DESCRIPTION = "Simple Sensor Monitor Service"

LICENSE = "MIT"

LIC_FILES_CHKSUM = "file://${COMMON_LICENSE_DIR}/MIT;md5=0835ade698e0bcf8506ecda2f7b4f302"

SRC_URI = " \
    file://sensor-monitor.cpp \
    file://sensor-monitor.service \
"

S = "${WORKDIR}"

inherit systemd

SYSTEMD_SERVICE:${PN} = "sensor-monitor.service"

do_compile() {
    ${CXX} ${CXXFLAGS} sensor-monitor.cpp -o sensor-monitor
}

do_install() {

    install -d ${D}${bindir}

    install -m 0755 sensor-monitor \
        ${D}${bindir}/sensor-monitor

    install -d ${D}${systemd_system_unitdir}

    install -m 0644 \
        ${WORKDIR}/sensor-monitor.service \
        ${D}${systemd_system_unitdir}/sensor-monitor.service
}

FILES:${PN} += " \
    ${bindir}/sensor-monitor \
    ${systemd_system_unitdir}/sensor-monitor.service \
"

EOF


# ============================================================
# STEP 10 — Add Package Into Image
# ============================================================

echo 'IMAGE_INSTALL:append = " sensor-monitor"' >> conf/local.conf


# ============================================================
# STEP 11 — Build Image
# ============================================================


DISTRO_FEATURES:append = " systemd"

VIRTUAL-RUNTIME_init_manager = "systemd"

VIRTUAL-RUNTIME_initscripts = ""

INIT_MANAGER = "systemd"



bitbake core-image-minimal


# ============================================================
# STEP 12 — Run QEMU
# ============================================================

runqemu qemuarm64 nographic


# ============================================================
# STEP 13 — Inside QEMU Verify Service
# ============================================================

systemctl status sensor-monitor


# ============================================================
# STEP 14 — Check Logs
# ============================================================

journalctl -u sensor-monitor


# ============================================================
# EXPECTED OUTPUT
# ============================================================

Sensor monitor running
Sensor monitor running
Sensor monitor running


# ============================================================
# OPTIONAL DEBUG COMMANDS
# ============================================================

# Check binary
which sensor-monitor

# Check process
ps -ef | grep sensor-monitor

# Check installed service
ls /lib/systemd/system/sensor-monitor.service

# Restart service
systemctl restart sensor-monitor

# Follow logs live
journalctl -fu sensor-monitor
```
